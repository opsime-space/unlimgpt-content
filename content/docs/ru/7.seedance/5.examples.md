Сохраните одно из тел ниже в `request.json`. Задайте `BASE_URL` и `API_KEY` по инструкции [Подключение](/ru/docs/seedance/connect). Каждый POST может создать платную задачу; сохраните полученный ID и следуйте разделу [Результаты](/ru/docs/seedance/results) для статуса и скачивания.

## Текст в видео

```json
{
  "model": "seedance-2.5",
  "generate_audio": false,
  "prompt": "A paper boat glides across a quiet lake at sunrise, one continuous shot.",
  "seconds": "5",
  "size": "1280x720"
}
```

```ts
const root = new URL(Bun.env.BASE_URL ?? "");
const key = Bun.env.API_KEY;
const taskPath = "/v1/videos";
if (
  root.protocol !== "https:" ||
  root.pathname !== "/" ||
  root.username ||
  root.password ||
  root.search ||
  root.hash ||
  !key

  throw new Error("Set HTTPS BASE_URL and API_KEY");

const response = await fetch(root.origin + taskPath, {
  method: "POST",
  redirect: "manual",
  headers: {

    "Content-Type": "application/json",
  },
  body: await Bun.file("request.json").text(),
});
if (!response.ok) {
  throw new Error(
    `Create HTTP ${response.status}; check account before another POST`,
  );
}
const task = await response.json() as { id?: string };
if (!task.id) {
  throw new Error("Missing task ID; check account before another POST");
}
await Bun.write("seedance-task.json", JSON.stringify({ id: task.id }));
console.log(task.id);
```

#python
```python
# Python 3: python3 create.py
import json
import os
import urllib.parse
import urllib.request
root = os.environ["BASE_URL"]
key = os.environ["API_KEY"]
origin = urllib.parse.urlsplit(root)
if (
    origin.scheme != "https" or not origin.netloc or origin.path
    or origin.query or origin.fragment or origin.username or origin.password
    or not key
):
    raise ValueError("Set HTTPS BASE_URL and API_KEY")


class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, *args):
        return None


opener = urllib.request.build_opener(NoRedirect)
with open("request.json", "rb") as file:
    body = file.read()
request = urllib.request.Request(
    root + "/v1/videos", data=body,
    headers={"Authorization": "Bearer " + key, "Content-Type": "application/json"},
    method="POST",
)
with opener.open(request, timeout=120) as response:
    task = json.load(response)
if not task.get("id"):
    raise RuntimeError("Missing task ID; check account before another POST")
with open("seedance-task.json", "w", encoding="utf8") as file:
    json.dump({"id": task["id"]}, file)
print(task["id"])
```
#go
```go
// Go 1.22+: go run create.go
package main
import ("bytes"; "encoding/json"; "fmt"; "net/http"; "os"; "time")
func main() {
  body, err := os.ReadFile("request.json"); if err != nil { panic(err) }
  req, err := http.NewRequest("POST", os.Getenv("BASE_URL")+"/v1/videos", bytes.NewReader(body)); if err != nil { panic(err) }
  req.Header.Set("Authorization", "Bearer "+os.Getenv("API_KEY"))
  req.Header.Set("Content-Type", "application/json")
  client := &http.Client{Timeout:120*time.Second, CheckRedirect:func(*http.Request, []*http.Request) error{return http.ErrUseLastResponse}}
  res, err := client.Do(req); if err != nil { panic(err) }; defer res.Body.Close()
  if res.StatusCode != 200 { panic(fmt.Sprintf("Create HTTP %d; check account before another POST", res.StatusCode)) }
  var task struct{ID string `json:"id"`}
  if err := json.NewDecoder(res.Body).Decode(&task); err != nil { panic(err) }
  if task.ID == "" { panic("Missing task ID; check account before another POST") }
  saved, _ := json.Marshal(struct{ID string `json:"id"`}{task.ID})
  if err := os.WriteFile("seedance-task.json", saved, 0600); err != nil { panic(err) }
  fmt.Println(task.ID)
}
```

## Создание с вебхуком

Добавьте адрес своего публичного HTTPS-обработчика в тот же `request.json`:

```json
{
  "model": "seedance-2.5",
  "resolution": "720p",
  "generate_audio": false,
  "callback_url": "CALLBACK_URL",
  "prompt": "A paper boat glides across a quiet lake.",
  "seconds": "5"
}
```

Запрос создания по-прежнему возвращает ID. Итог приходит отдельным POST на `callback_url`; [формат уведомления](/ru/docs/seedance/results) и [порядок обработки](/ru/docs/seedance/integration) описаны отдельно.

## Сценарии JSON

Видео с последним кадром результата в JPEG:

```json
{
  "model": "seedance-2.0",
  "resolution": "720p",
  "generate_audio": false,
  "return_last_frame": true,
  "prompt": "A ceramic cup rotates slowly on a table, stable lighting.",
  "seconds": "5",
  "aspect_ratio": "1:1"
}
```

После завершения используйте `content.last_frame_url` из [ответа задачи](/ru/docs/seedance/results). Флаг можно добавить и в запрос с `callback_url`.

Портрет со звуком:

```json
{
  "model": "seedance-2.5",
  "generate_audio": true,
  "prompt": "A cyclist crosses a rainy street at dusk. Gentle tire sounds; one continuous shot.",
  "seconds": "10",
  "size": "720x1280"
}
```

Квадрат 1080p:

```json
{
  "model": "seedance-2.0",
  "resolution": "1080p",
  "prompt": "A ceramic cup rotates slowly on a table, stable lighting.",
  "seconds": "12",
  "aspect_ratio": "1:1"
}
```

Автоматическая длительность только у 2.5:

```json
{
  "model": "seedance-2.5",
  "resolution": "720p",
  "prompt": "A small kite rises above a beach.",
  "seconds": "-1"
}
```

Mini и Fast требуют текст. Если добавлен аудиореференс, им также нужно изображение или видео. Роли медиа, смешанные референсы, edit, extend и черновик описаны в [Референсах](/ru/docs/seedance/references). Доступность режимов зависит от подключения и ключа.
