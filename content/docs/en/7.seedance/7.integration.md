Use [Connect](/docs/seedance/connect) and save a JSON body from [Examples](/docs/seedance/examples) as `request.json`. Keep `BASE_URL` as an HTTPS origin and the key on your server. One state file belongs to one order. Rerunning a program with the saved ID resumes GET without another POST.

## Webhook integration

Add `callback_url` to `request.json` as shown in the
[creation example](/docs/seedance/examples). The cURL, TypeScript, Python and
Go submission examples use that JSON with the same method and path.

Create the task once and save its ID. Your HTTPS handler accepts a POST with
the [task snapshot](/docs/seedance/results), persists it and returns
`200` or `204` promptly. Process it after acknowledging: read the task with
an authenticated GET, check its status and download the result. Do not download
the video before acknowledging the webhook.

Keep a periodic GET every 60–120 seconds to recover missed notifications.
A repeated callback does not create another order and must not repeat your
application's actions. When `callback_url` is omitted, use the normal polling
workflow below.

```bash
umask 077
touch result.json content.headers && chmod 600 result.json content.headers || exit 1
[[ "$BASE_URL" =~ ^https://[A-Za-z0-9.-]+(:[0-9]{1,5})?$ &&
  -n "$API_KEY" && -r request.json ]] || {
  printf 'Set HTTPS BASE_URL, API_KEY and request.json\n' >&2
  exit 1
}
OUTPUT_FILE=${VIDEO_OUTPUT:-seedance.mp4}
if [[ -f task-id.txt ]]; then
  TASK_ID=$(< task-id.txt)
else
  (
    set -C
    printf 'creating\n' > task-id.txt
  ) || {
    printf 'State file already exists; inspect it before another POST\n' >&2
    exit 1
  }
  if curl --fail --silent --show-error --max-time 120 \
    "$BASE_URL/v1/videos" \
    -H "Authorization: Bearer $API_KEY" \
    -H 'Content-Type: application/json' \
    --data-binary @request.json \
    -o created.json; then
    TASK_ID=$(jq -er '
      .id | select(type == "string" and length > 0)
    ' created.json) || exit 1
    printf '%s\n' "$TASK_ID" > task-id.txt
  else
    printf 'Outcome unknown; check console before another POST\n' >&2
    exit 1
  fi
fi
[[ "$TASK_ID" != creating && "$TASK_ID" =~ ^[A-Za-z0-9_-]+$ ]] || {
  printf 'Invalid task ID\n' >&2
  exit 1
}
deadline=$((SECONDS + 2700))
completed=0
while ((SECONDS < deadline)); do
  if ! http_status=$(curl --silent --show-error --max-time 60 \
    --write-out '%{http_code}' \
    "$BASE_URL/v1/videos/$TASK_ID" \
    -H "Authorization: Bearer $API_KEY" \
    -o result.json); then
    sleep 30
    continue
  fi
  case "$http_status" in
    200) ;;
    429 | 5??)
      sleep 30
      continue
      ;;
    *)
      printf 'Status HTTP %s; ID preserved\n' "$http_status" >&2
      exit 1
      ;;
  esac
  jq '{id, status, error}' result.json
  status=$(jq -r '.status' result.json)
  case "$status" in
    completed)
      completed=1
      break
      ;;
    failed)
      printf 'Terminal status: %s\n' "$status" >&2
      exit 1
      ;;
    queued | in_progress) sleep 15 ;;
    *)
      printf 'Unknown status: %s\n' "$status" >&2
      exit 1
      ;;
  esac
done
if ((!completed)); then
  printf 'Polling limit reached; resume with %s\n' "$TASK_ID" >&2
  exit 1
fi
VIDEO_URL=$(jq -r '.content.video_url // empty' result.json)
if [[ -z "$VIDEO_URL" ]]; then
  metadata=$(curl --silent --show-error --max-time 300 \
    --dump-header content.headers --write-out '%{http_code} %{content_type}' \
    "$BASE_URL/v1/videos/$TASK_ID/content" \
    -H "Authorization: Bearer $API_KEY" -o seedance.download) || exit 1
  http_status=${metadata%% *}
  if [[ "$http_status" == 307 ]]; then
    VIDEO_URL=$(awk 'tolower($1)=="location:" {print $2}' content.headers | tr -d '\r' | tail -1)
  elif [[ "$http_status" == 200 ]]; then
    mime=${metadata#* }
    case "$mime" in
      video/mp4* | video/quicktime* | application/octet-stream*)
        mv seedance.download "$OUTPUT_FILE"
        exit 0 ;;
    esac
  fi
  [[ "$http_status" == 307 ]] || { printf 'Content HTTP %s; ID preserved\n' "$http_status" >&2; exit 1; }
fi
[[ "$VIDEO_URL" =~ ^https://[^/@?#]+ ]] &&
  [[ ! "$VIDEO_URL" =~ ^https://[^/?#]*@ ]] || {
  printf 'Invalid signed HTTPS media URL\n' >&2
  exit 1
}
# The signed URL grants temporary access. Never send the API Bearer to it.
metadata=$(curl --silent --show-error --max-time 300 \
  --write-out '%{http_code} %{content_type}' \
  "$VIDEO_URL" -o seedance.download) || exit 1
http_status=${metadata%% *}
mime=${metadata#* }
case "$http_status:$mime" in
  2??:video/mp4* | 2??:video/quicktime* | 2??:application/octet-stream*)
    mv seedance.download "$OUTPUT_FILE"
    ;;
  *)
    printf 'Unexpected HTTP response or file type\n' >&2
    exit 1
    ;;
esac
```

```ts
// Bun 1.4+: bun run seedance.ts
const root = new URL(Bun.env.BASE_URL ?? "");
const key = Bun.env.API_KEY;
if (root.protocol !== "https:" || root.pathname !== "/" || root.username ||
    root.password || root.search || root.hash || !key) {
  throw new Error("Set HTTPS BASE_URL and API_KEY");
}
const state = Bun.env.TASK_STATE ?? "seedance-task.json";
const output = Bun.env.VIDEO_OUTPUT ?? "seedance.mp4";
const taskPath = "/v1/videos";
const api = (path: string, init: RequestInit = {}) =>
  fetch(root.origin + path, {
    ...init,
    headers: { Authorization: `Bearer ${key}`, ...init.headers },
    redirect: "manual",
    signal: init.signal ?? AbortSignal.timeout(120_000),
  });
let id: string;
if (await Bun.file(state).exists()) {
  const saved = await Bun.file(state).json() as { id?: string };
  if (!saved.id || !/^[A-Za-z0-9_-]+$/.test(saved.id)) {
    throw new Error("Invalid task state; inspect it before another POST");
  }
  id = saved.id;
} else {
  // Reserve this state filename for one order before POST. If creation is
  // interrupted, inspect the account before deciding whether to create again.
  await Bun.write(state, JSON.stringify({ creating: true }));
  const created = await api(taskPath, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: await Bun.file("request.json").text(),
  });
  if (!created.ok) throw new Error(`Create HTTP ${created.status}; check account before another POST`);
  const receipt = await created.json() as { id?: string };
  if (!receipt.id || !/^[A-Za-z0-9_-]+$/.test(receipt.id)) {
    throw new Error("Task ID missing; check account before another POST");
  }
  id = receipt.id;
  await Bun.write(state, JSON.stringify({ id }));
  console.log(`Task ID: ${id}`);
}
const deadline = Date.now() + 45 * 60_000;
let videoUrl: string | undefined;
let completed = false;
while (Date.now() < deadline) {
  let response: Response;
  try {
    response = await api(`${taskPath}/${encodeURIComponent(id)}`);
  } catch {
    await Bun.sleep(15_000);
    continue;
  }
  if (response.status === 429 || response.status >= 500) {
    await response.body?.cancel();
    await Bun.sleep(15_000);
    continue;
  }
  if (!response.ok) throw new Error(`Status HTTP ${response.status}; ID preserved`);
  const task = await response.json() as { status?: string; content?: { video_url?: string } };
  if (task.status === "completed") {
    videoUrl = task.content?.video_url;
    completed = true;
    break;
  }
  if (task.status === "failed") throw new Error(`Task failed; ID ${id} preserved`);
  if (task.status !== "queued" && task.status !== "in_progress") {
    throw new Error(`Unknown task status; ID ${id} preserved`);
  }
  await Bun.sleep(15_000);
}
if (!completed) throw new Error(`Polling timed out; resume with ${id}`);
let video: Response | undefined;
if (!videoUrl) {
  const content = await api(`${taskPath}/${encodeURIComponent(id)}/content`);
  if (content.status === 307) {
    videoUrl = content.headers.get("location") ?? undefined;
    await content.body?.cancel();
  } else if (content.ok) {
    video = content;
  } else {
    throw new Error(`Content HTTP ${content.status}; ID preserved`);
  }
}
if (videoUrl) {
  const media = new URL(videoUrl);
  if (media.protocol !== "https:" || !media.hostname || media.username || media.password) {
    throw new Error("Invalid signed HTTPS media URL");
  }
  // This signed URL is a temporary credential. Never forward the API Bearer.
  video = await fetch(media.href, { redirect: "manual", signal: AbortSignal.timeout(300_000) });
}
if (!video) throw new Error(`Content URL missing; ID ${id} preserved`);
if (!video.ok) throw new Error(`Download HTTP ${video.status}; re-read task ${id}, then contact support`);
const mime = video.headers.get("content-type")?.split(";")[0].trim().toLowerCase();
if (!["video/mp4", "video/quicktime", "application/octet-stream"].includes(mime ?? "")) {
  throw new Error("Unexpected download content type");
}
await Bun.write(output, new Uint8Array(await video.arrayBuffer()));
```

```python
# Python 3: python3 seedance.py
import json
import os
import re
import time
import urllib.error
import urllib.parse
import urllib.request

root, key = os.environ["BASE_URL"], os.environ["API_KEY"]
origin = urllib.parse.urlsplit(root)
if (
    origin.scheme != "https"
    or not origin.netloc
    or origin.path
    or origin.query
    or origin.fragment
    or origin.username
    or origin.password
):
    raise ValueError("BASE_URL must be an HTTPS origin")
state, output = (
    os.getenv("TASK_STATE", "seedance-task.json"),
    os.getenv("VIDEO_OUTPUT", "seedance.mp4"),
)


class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, request, fp, code, msg, headers, newurl):
        return None


opener = urllib.request.build_opener(NoRedirect)


def call(path, data=None, timeout=120):
    headers = {"Authorization": "Bearer " + key}
    if data is not None:
        headers["Content-Type"] = "application/json"
    request = urllib.request.Request(
        root + path,
        data=data,
        headers=headers,
        method="POST" if data is not None else "GET",
    )
    return opener.open(request, timeout=timeout)


try:
    with open(state, encoding="utf8") as file:
        task_id = json.load(file)["id"]
except FileNotFoundError:
    with open("request.json", "rb") as file:
        body = file.read()
    with open(state, "x", encoding="utf8") as file:
        json.dump({"creating": True}, file)
    with call("/v1/videos", body) as response:
        receipt = json.load(response)
    task_id = receipt.get("id")
    if not isinstance(task_id, str) or not re.fullmatch(
        r"[A-Za-z0-9_-]+", task_id
    ):
        raise RuntimeError("Task ID missing; check console before another POST")
    print("Task ID:", task_id)
    with open(state, "w", encoding="utf8") as file:
        json.dump({"id": task_id}, file)
if not isinstance(task_id, str) or not re.fullmatch(r"[A-Za-z0-9_-]+", task_id):
    raise ValueError("Invalid task state")
deadline, video_path, downloaded = time.monotonic() + 45 * 60, None, False

while time.monotonic() < deadline:
    try:
        with call("/v1/videos/" + task_id, timeout=60) as response:
            task = json.load(response)
    except urllib.error.HTTPError as error:
        if error.code != 429 and error.code < 500:
            raise
        time.sleep(30)
        continue
    except (urllib.error.URLError, TimeoutError):
        time.sleep(15)
        continue
    if task.get("status") == "completed":
        video_url = task.get("content", {}).get("video_url")
        if isinstance(video_url, str):
            target = urllib.parse.urlsplit(video_url)
            if target.scheme != "https" or not target.netloc or target.username or target.password:
                raise ValueError("Invalid signed HTTPS media URL")
            video_path = video_url
        else:
            try:
                content = call("/v1/videos/" + task_id + "/content", timeout=300)
            except urllib.error.HTTPError as error:
                content = error
            with content:
                if content.status == 307:
                    video_path = content.headers.get("Location")
                    target = urllib.parse.urlsplit(video_path or "")
                    if target.scheme != "https" or not target.netloc or target.username or target.password:
                        raise ValueError("Invalid signed HTTPS media URL")
                elif content.status == 200:
                    if content.headers.get_content_type() not in ("video/mp4", "video/quicktime", "application/octet-stream"):
                        raise RuntimeError("Unexpected content type")
                    with open(output, "wb") as file:
                        file.write(content.read())
                    downloaded = True
                else:
                    raise RuntimeError("Content HTTP %s; ID preserved" % content.status)
        break
    if task.get("status") in ("failed",):
        raise RuntimeError("Terminal task status")
    if task.get("status") not in ("queued", "in_progress"):
        raise RuntimeError("Unknown task status")
    time.sleep(15)
if downloaded:
    raise SystemExit(0)
if not video_path:
    raise TimeoutError("Polling timed out; resume with saved ID")
with opener.open(urllib.request.Request(video_path), timeout=300) as response:
    if response.headers.get_content_type() not in (
        "video/mp4",
        "video/quicktime",
        "application/octet-stream",
    ):
        raise RuntimeError("Unexpected download content type")
    with open(output, "wb") as file:
        while chunk := response.read(1024 * 1024):
            file.write(chunk)
```

#go
```go
// Go 1.22+: go run seedance.go
package main

import (
	"bytes"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"net/http"
	"net/url"
	"os"
	"regexp"
	"strings"
	"time"
)

type Task struct {
	ID      string `json:"id"`
	Status  string `json:"status"`
	Content struct {
		VideoURL string `json:"video_url"`
	} `json:"content"`
}

func main() {
	root, err := url.Parse(os.Getenv("BASE_URL"))
	if err != nil ||
		root == nil ||
		root.Scheme != "https" ||
		root.Host == "" ||
		root.User != nil ||
		root.Path != "" ||
		root.RawQuery != "" ||
		root.Fragment != "" {
		panic("BASE_URL must be an HTTPS origin")
	}
	key := os.Getenv("API_KEY")
	if key == "" {
		panic("Set API_KEY")
	}
	state := os.Getenv("TASK_STATE")
	if state == "" {
		state = "seedance-task.json"
	}
	output := os.Getenv("VIDEO_OUTPUT")
	if output == "" {
		output = "seedance.mp4"
	}
	client := &http.Client{
		Timeout: 120 * time.Second,
		CheckRedirect: func(*http.Request, []*http.Request) error {
			return http.ErrUseLastResponse
		},
	}
	call := func(path, method string, body io.Reader) (*http.Response, error) {
		req, err := http.NewRequest(method, root.String()+path, body)
		if err != nil {
			return nil, err
		}
		req.Header.Set("Authorization", "Bearer "+key)
		if method == "POST" {
			req.Header.Set("Content-Type", "application/json")
		}
		return client.Do(req)
	}
	var id string
	saved, err := os.ReadFile(state)
	if err == nil {
		var task Task
		if err := json.Unmarshal(saved, &task); err != nil {
			panic(err)
		}
		id = task.ID
	} else if errors.Is(err, os.ErrNotExist) {
		body, err := os.ReadFile("request.json")
		if err != nil {
			panic(err)
		}

		file, err := os.OpenFile(state, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
		if err != nil {
			panic(err)
		}
		if _, err := file.WriteString("{\"creating\":true}\n"); err != nil {
			file.Close()
			panic(err)
		}
		if err := file.Close(); err != nil {
			panic(err)
		}
		response, err := call("/v1/videos", "POST", bytes.NewReader(body))
		if err != nil {
			panic(err)
		}
		if response.StatusCode != 200 {
			response.Body.Close()
			panic(fmt.Sprintf(
				"Create HTTP %d; check console before another POST",
				response.StatusCode,
			))
		}

		var receipt Task
		err = json.NewDecoder(response.Body).Decode(&receipt)
		response.Body.Close()
		if err != nil {
			panic(err)
		}
		id = receipt.ID
		if !regexp.MustCompile(`^[A-Za-z0-9_-]+$`).MatchString(id) {
			panic("Task ID missing; check console before another POST")
		}

		fmt.Println("Task ID:", id)
		saved, err := json.Marshal(Task{ID: id})
		if err != nil {
			panic(err)
		}

		if err := os.WriteFile(state, saved, 0600); err != nil {
			panic(err)
		}
	} else {
		panic(err)
	}
	if !regexp.MustCompile(`^[A-Za-z0-9_-]+$`).MatchString(id) {
		panic("Invalid task state")
	}

	deadline := time.Now().Add(45 * time.Minute)
	var path string
	var completed bool

	for time.Now().Before(deadline) {
		response, err := call("/v1/videos/"+url.PathEscape(id), "GET", nil)
		if err != nil {
			time.Sleep(15 * time.Second)
			continue
		}
		if response.StatusCode == 429 || response.StatusCode >= 500 {
			response.Body.Close()
			time.Sleep(30 * time.Second)
			continue
		}
		if response.StatusCode != 200 {
			response.Body.Close()
			panic(fmt.Sprintf("Status HTTP %d", response.StatusCode))
		}

		var task Task
		err = json.NewDecoder(response.Body).Decode(&task)
		response.Body.Close()
		if err != nil {
			panic(err)
		}
		if task.Status == "completed" {
			path = task.Content.VideoURL
			completed = true
			break
		}
		if task.Status == "failed" {
			panic("Terminal task status: " + task.Status)
		}
		if task.Status != "queued" && task.Status != "in_progress" {
			panic("Unknown task status")
		}
		time.Sleep(15 * time.Second)
	}
	if !completed {
		panic("Polling timed out; resume with saved ID")
	}

	client.Timeout = 300 * time.Second
	var response *http.Response
	if path == "" {
		response, err = call("/v1/videos/"+url.PathEscape(id)+"/content", "GET", nil)
		if err != nil { panic(err) }
		if response.StatusCode == 307 {
			path = response.Header.Get("Location")
			response.Body.Close()
		} else if response.StatusCode != 200 {
			response.Body.Close()
			panic(fmt.Sprintf("Content HTTP %d; ID preserved", response.StatusCode))
		}
	}
	if path != "" {
		target, err := url.Parse(path)
		if err != nil || target.Scheme != "https" || target.Host == "" || target.User != nil {
			panic("Invalid signed HTTPS media URL")
		}
		request, err := http.NewRequest("GET", target.String(), nil)
		if err != nil { panic(err) }
		response, err = client.Do(request)
		if err != nil { panic(err) }
	}
	if response == nil { panic("Content response missing; ID preserved") }
	defer response.Body.Close()
	if response.StatusCode != 200 {
		panic(fmt.Sprintf("Download HTTP %d", response.StatusCode))
	}
	mime := response.Header.Get("Content-Type")
	if !strings.HasPrefix(mime, "video/") && !strings.HasPrefix(mime, "application/octet-stream") {
		panic("Unexpected download content type")
	}

	file, err := os.Create(output)
	if err != nil {
		panic(err)
	}
	defer file.Close()
	if _, err := io.Copy(file, response.Body); err != nil {
		panic(err)
	}

	fmt.Println("Saved", output)
}
```

::

Run the cURL steps in sequence, repeating only status GET every 10-15 seconds until a terminal status or a local 45-minute deadline. On a POST timeout before an ID arrives, check the console or support before another POST. For a new order use a new state filename. A local deadline does not cancel the remote task. For MOV output set `VIDEO_OUTPUT=seedance.mov` (all examples use this filename). Protect local state and result files because result JSON may contain a signed media URL.
